.text
.globl main

main:
    la t0, matriz          # t0 = dirección base
    li t1, 0               # t1 = suma = 0
    li t2, 0               # t2 = índice global

    li t3, FILAS*COLS      # número total de elementos

loop:
    beq t2, t3, fin        # si recorrimos todos → fin
    lw t4, 0(t0)           # cargar elemento
    add t1, t1, t4         # suma += elemento
    addi t0, t0, 4         # mover al siguiente
    addi t2, t2, 1
    j loop

fin:
    # en t1 queda la suma de todos los elementos
    li a7, 10
    ecall

.data
matriz:
    .word 1, 2, 3,
          4, 5, 6

.equ FILAS, 2
.equ COLS, 3
