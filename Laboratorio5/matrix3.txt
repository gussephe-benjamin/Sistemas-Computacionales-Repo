.text
.globl main

main
    la t0, matriz          # t0 = dirección base
    li t1, 1               # fila = 1
    li t2, 2               # columna = 2

    # Calcular índice = (fila  COLS) + columna
    mul t3, t1, COLS       # t3 = fila  COLS
    add t3, t3, t2         # t3 = t3 + columna

    # Calcular desplazamiento en bytes = índice  4
    slli t3, t3, 2         # t3 = t3  4

    # Dirección del elemento
    add t0, t0, t3

    # Cambiar valor
    li t4, 99              # nuevo valor
    sw t4, 0(t0)           # guardar en memoria

    li a7, 10              # salir
    ecall

.data
matriz
    .word 1, 2, 3,
          4, 5, 6

.equ FILAS, 2
.equ COLS, 3
